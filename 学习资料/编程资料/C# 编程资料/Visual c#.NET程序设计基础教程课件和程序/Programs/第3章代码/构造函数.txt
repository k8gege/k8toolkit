    public class MyBaseClass
    {
        //构造函数：用于创建对象使用。
        //构造函数特性：
        //1。Obect 提供一个默认的构造函数，这个构造函数没有任何参数。
        //2。构造函数和该类同名
        //3。构造函数 不需要指定返回类型，或者void ，它和 new 关键字搭配使用用于创建一个该类型的对象。
        //4。声明有其他带有参数的构造函数后，默认的构造函数不能再使用，除非特别再声明该默认的构造函数
        public MyBaseClass()
        {
            Console.WriteLine("执行 MyBaseClass.MyBaseClass() 构造函数 （默认的构造函数）");
        }

        //函数的重载：在同一个类里面，函数名称相同，但是参数不同。

        public MyBaseClass(int i)
        {
            Console.WriteLine("执行 MyBaseClass.MyBaseClass(int i) 构造函数！");
        }

      
    }
    public class MyDerivedClass:MyBaseClass
    {
        public MyDerivedClass()
        {
            Console.WriteLine("执行 MyDerivedClass.MyDerivedClass() 构造函数 （默认的构造函数）");
        }
        public MyDerivedClass(int i)
        {
            Console.WriteLine("执行 MyDerivedClass.MyDerivedClass(int i) 构造函数！");
        }
        public MyDerivedClass(int i,int j):this(i)
        {
            Console.WriteLine("执行 MyDerivedClass.MyDerivedClass(int i,int j) 构造函数！");
        }

        //使用 static 来指定该 方法是静态的方法
        // 不能使用实例来调用该方法；
        public static void Test()
        {
            
        }

        // 该方法是一个实例方法
        public void Test2()
        {
            
        }


        ~MyDerivedClass()
        { 
            
        }
    
    }
