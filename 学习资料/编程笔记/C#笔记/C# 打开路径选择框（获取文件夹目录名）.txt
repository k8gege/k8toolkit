使用FolderBrowserDialog，.net   Framework1.1中新增的类：  

按钮处理函数用来打开一个路径选择框，


        private void btn_dirpath_Click(object sender, EventArgs e)
        {
            //From:QQ吻's Blog http://qqhack8.blog.163.com
            FolderBrowserDialog myDialog = new FolderBrowserDialog();

            myDialog.ShowNewFolderButton = false;

            myDialog.Description = "选择你要设置的文件夹或目录 ";

            if (myDialog.ShowDialog() == DialogResult.OK)

                this.txtDirPath.Text = myDialog.SelectedPath;

            myDialog.Dispose();
        }
